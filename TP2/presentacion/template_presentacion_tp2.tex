% 3.Modelos matemáticos
\begin{frame}{Modelos}
  Posición y velocidad (común a los dos modelos):
  \begin{align*}
    \vec{r}_i(t+\Delta t) &= \vec{r}_i(t) + \vec{v}_i(t)\Delta t,\\
    \vec{v}_i(t) &= v_0\big(\cos\theta_i(t), \sin\theta_i(t)\big)
  \end{align*}
  \vspace{0.05cm}
  \textbf{Modelo estándar} --- actualización angular:
  \[
    \theta_i(t+\Delta t) = \langle\theta_j(t)\rangle_{j\in\mathcal{V}_i(t)} + \Delta\theta_i(t)
  \]
  \textbf{Modelo votante} --- actualización angular:
  \[
    \theta_i(t+\Delta t) = \theta_j(t) + \Delta\theta_i(t), \; j\sim U(\mathcal{V}_i(t))
  \]
  {\scriptsize Si $\mathcal{V}_i(t)=\emptyset$: $\theta_i(t+\Delta t) = \theta_i(t) + \Delta\theta_i(t)$ (ambos modelos).\par}
  \vspace{0.05cm}
  Ruido (común a los dos modelos):
  \[
    \Delta\theta_i(t) \sim U(-\eta/2,\ \eta/2)
  \]
\end{frame}

% 10. Observable principal
\begin{frame}{Observable Definido}
  \centering
  Parámetro de orden $v_a(t)$
  \begin{equation*}
    v_a=\frac{1}{N v_0}\left|\sum_{i=1}^{N}\bm{v}_i\right|
  \end{equation*}
  \begin{itemize}
    \item Valor alto: alineamiento colectivo (orden).
    \item Valor bajo: movimiento desordenado, sin dirección preferencial.
  \end{itemize}
  Comparado entre modelos y densidades, promediando $M=5$ corridas en estado estacionario.
\end{frame}

\begin{frame}{Conclusiones}
  Principales conclusiones obtenidas:
  \begin{itemize}
    \item Hallazgo principal: modelo estándar tolera mucho más ruido antes de perder
      orden colectivo que el votante.
    \item Parámetro principal: la densidad afecta bastante al estándar, casi nada al votante.
    \item Comparación: promedio vectorial (estándar) sistemáticamente más resistente
      al ruido que copia de un vecino al azar (votante).
    \item Trabajo futuro: profundizar el estudio de clusters ($S$) en el modelo votante,
      no incluido en este TP.
  \end{itemize}
\end{frame}
