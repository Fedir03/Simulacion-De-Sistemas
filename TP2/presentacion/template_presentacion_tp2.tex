\newcommand{\TPnavdot}[2]{%
  \makebox[0.14cm][c]{%
    \raisebox{0.05ex}{\textcolor{#1}{\fontsize{6}{6}\selectfont #2}}}%
}

\newcommand{\navblock}[5]{%
  % #1 ancho, #2 nombre, #3 inicio, #4 cantidad, #5 fin
  \begin{minipage}[c][0.82cm][c]{#1\paperwidth}
    \centering
    \ifnum\insertframenumber>1
      \ifnum\insertframenumber<#3
        {\color{TPNavGray}\scriptsize #2\par}
      \else
        \ifnum\insertframenumber>#5
          {\color{TPNavGray}\scriptsize #2\par}
        \else
          {\color{white}\scriptsize #2\par}
        \fi
      \fi
    \else
      {\color{TPNavGray}\scriptsize #2\par}
    \fi
    \vspace{0.03cm}
    \@tempcnta=0
    \loop
      \advance\@tempcnta by 1
      \@tempcntb=#3
      \advance\@tempcntb by \@tempcnta
      \advance\@tempcntb by -1
      \ifnum\insertframenumber<#3
        \TPnavdot{TPNavDotGray}{\textopenbullet}%
      \else
        \ifnum\insertframenumber>#5
          \TPnavdot{TPNavDotGray}{\textbullet}%
        \else
          \ifnum\insertframenumber<\@tempcntb
            \TPnavdot{white}{\textopenbullet}%
          \else
            \TPnavdot{white}{\textbullet}%
          \fi
        \fi
      \fi
    \ifnum\@tempcnta<#4
    \repeat
  \end{minipage}%
}
